    bonus = 0

    // СИНТАКСИЧЕСКАЯ ОШИБКА: Пропущено двоеточие ':' после условия

        bonus = 10000
    else :
        bonus = 5000
    end if

    // (Анализатор до нее не дойдет, пока не исправим двоеточие выше)

end sub
